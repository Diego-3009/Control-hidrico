Integración de Python en el Ecosistema

Aunque el microcontrolador principal ejecuta lógica embebida, herramientas de software complementarias en Python permiten procesar datos telemétricos, simular respuestas lógicas de los componentes y gestionar interfaces de control auxiliar conectadas al sistema hídrico.

Estructura Lógica de Control (Simulador Python)

Ejemplo de script utilizado para la evaluación de rangos de distancia y activación automática de alertas:

# Simulación de control de nivel de agua - HydroControl Dynamics
#include 

// Inicializa el LCD (direccion I2C 0x27, 16 columnas, 2 filas)
// Si la pantalla enciende pero no muestra texto, prueba con 0x3F
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Pines del sensor ultrasonico
const int echoPin = 3;
const int trigPin = 4;

// Pines de los LEDs
const int greenLed = 7;  // LED verde en el pin 7
const int blueLed = 6;   // LED azul en el pin 6
const int redLed = 5;    // LED rojo en el pin 5

void setup() {
  // Inicializa el LCD
  lcd.init();
  lcd.clear();
  lcd.backlight();

  // Pines del sensor ultrasonico
  pinMode(echoPin, INPUT);
  pinMode(trigPin, OUTPUT);

  // Pines de los LEDs
  pinMode(greenLed, OUTPUT);
  pinMode(blueLed, OUTPUT);
  pinMode(redLed, OUTPUT);
}

void loop() {
  // Obtiene la distancia hasta el agua
  float distance = readDistance();

  // Muestra la distancia en el LCD
  lcd.setCursor(0, 0);
  lcd.print("Distance:");
  lcd.setCursor(0, 1);
  lcd.print(distance);
  lcd.print(" cm  ");

  // Enciende el LED segun la distancia
  if (distance <= 5) {
    // 0-5 cm: LED verde
    digitalWrite(greenLed, HIGH);
    digitalWrite(blueLed, LOW);
    digitalWrite(redLed, LOW);
  } else if (distance > 5 && distance <= 10) {
    // 5-10 cm: LED azul
    digitalWrite(greenLed, LOW);
    digitalWrite(blueLed, HIGH);
    digitalWrite(redLed, LOW);
  } else if (distance > 10 && distance <= 15) {
    // 10-15 cm: LED rojo
    digitalWrite(greenLed, LOW);
    digitalWrite(blueLed, LOW);
    digitalWrite(redLed, HIGH);
  } else {
    // Mas de 15 cm: todos apagados
    digitalWrite(greenLed, LOW);
    digitalWrite(blueLed, LOW);
    digitalWrite(redLed, LOW);
  }

  delay(500);  // Espera 500 milisegundos
}

// Lee la distancia del sensor ultrasonico
float readDistance() {
  digitalWrite(trigPin, LOW);   // Asegura que el pin trig este en bajo
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);  // Envia el pulso ultrasonico
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);   // Termina el disparo

  // Mide el ancho del pulso en echo y lo convierte a cm
  float distance = pulseIn(echoPin, HIGH) / 58.00;
  return distance;
}
